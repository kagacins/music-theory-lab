# Progression Builder Integration Plan
## Making Chord Progression the Heart of Composition

---

## Vision: Chord-First Composition Workflow

Traditional notation software (MuseScore, Finale) starts with **notes** → then analyzes harmony.

**Your unique approach**: Start with **harmony** → auto-generate and intelligently suggest notes.

This is the killer feature that differentiates Music Theory Lab from all competitors.

## Current State Analysis

### ✅ What Works (Progression Builder Strengths)
1. **3D Scoring System**: Evaluates 600+ chord combinations
2. **Modal Interchange**: Automatic borrowed chord detection
3. **Voice Leading Analysis**: Shows smooth transitions
4. **Context Awareness**: ii-V-I detection, cadence recognition
5. **Customizable Weights**: User control over recommendation priorities
6. **Multi-step Recommendations**: Look-ahead suggestions
7. **Comprehensive Explorer**: Detailed filtering and analysis

### ❌ Current Limitations (Melody Notation Area)
1. **One-way sync**: Chord progression → notation (no reverse)
2. **Single melody line**: No bass clef auto-fill
3. **No voicing editor**: Can't adjust how chords are played
4. **Static rhythm**: Chords don't adapt to notation rhythm
5. **No real-time suggestions**: Can't see chord recommendations while composing melody
6. **Limited playback**: Chords and melody play separately

---

## Core Integration Principles

### 1. **Bi-Directional Synchronization**
```
Chord Progression ←→ Notation

User changes chord progression → Notation updates
User changes chord in notation → Progression updates
```

### 2. **Real-Time Assistance**
```
User composes melody → System suggests chord changes
User adds chord → System suggests bass voicing + melody notes
```

### 3. **Intelligent Auto-Fill**
```
Progression exists → Auto-generate professional arrangement
User can override any auto-generated element
```

```javascript
export class ProgressionNotationSync {
  constructor(progressionState, notationState) {
    this.progression = progressionState;
    this.notation = notationState;

    // Listen for changes in both directions
    this.setupBidirectionalSync();
  }

  setupBidirectionalSync() {
    // Progression → Notation
    this.progression.on('chordChanged', (measureIndex, newChord) => {
      this.updateNotationFromProgression(measureIndex, newChord);
    });

    // Notation → Progression
    this.notation.on('chordChanged', (measureIndex, newChord) => {
      this.updateProgressionFromNotation(measureIndex, newChord);
    });
  }

  updateNotationFromProgression(measureIndex, chord) {
    // Update bass clef with new chord voicing
    const bassVoicing = this.generateBassVoicing(chord);
    this.notation.updateBass(measureIndex, bassVoicing);

    // Optionally highlight melody notes that match chord
    this.notation.highlightChordTones(measureIndex, chord);
  }

  updateProgressionFromNotation(measureIndex, chord) {
    // Update progression builder when user edits chord in notation
    this.progression.updateChord(measureIndex, chord);

    // Trigger recommendation refresh
    this.progression.refreshRecommendations();
  }
}
```

#### 1.2 Smart Bass Auto-Fill

**Feature**: Automatically generate bass clef from chord progression with voice leading

**Algorithm**:
```javascript
// modules/integration/bassAutoFill.js

export function generateBassVoicing(chord, previousChord, voiceLeadingStrict = true) {
  const chordNotes = getChordNotes(chord.root, chord.type);

  if (!previousChord) {
    // First chord: use root in bass
    return {
      notes: [
        { pitch: `${chord.root}2`, duration: "whole" }
      ]
    };
  }

  // Voice leading: choose bass note closest to previous
  const previousBass = previousChord.bass.notes[0].pitch;
  const closestNote = findClosestNote(chordNotes, previousBass);

  // Add rhythm variation based on time signature
  const rhythm = generateBassRhythm(chord, timeSignature);

  return {
    notes: rhythm.map(duration => ({
      pitch: closestNote,
      duration: duration
    }))
  };
}

function generateBassRhythm(chord, timeSig) {
  // 4/4: whole note or quarter-quarter-half pattern
  if (timeSig.num === 4 && timeSig.denom === 4) {
    return Math.random() > 0.5
      ? ["whole"]
      : ["quarter", "quarter", "half"];
  }

  // 3/4: dotted half or quarter-quarter-quarter
  if (timeSig.num === 3 && timeSig.denom === 4) {
    return Math.random() > 0.5
      ? ["half."]
      : ["quarter", "quarter", "quarter"];
  }

  return ["whole"];
}
```

```javascript
// modules/integration/smartChordInsertion.js

export async function insertChordWithArrangement(measureIndex, chord, options = {}) {
  // 1. Add to progression
  progressionBuilder.insertChord(measureIndex, chord);

  // 2. Generate bass voicing
  const previousChord = progressionBuilder.getChord(measureIndex - 1);
  const bassVoicing = generateBassVoicing(chord, previousChord, options.voiceLeading);

  // 3. Update notation
  notation.updateMeasure(measureIndex, {
    chord: chord,
    bass: bassVoicing,
    chordSymbol: `${chord.root}${chord.type}`
  });

  // 4. Suggest melody notes (optional overlay)
  if (options.showMelodySuggestions) {
    const melodySuggestions = suggestMelodyNotes(chord, key);
    notation.showMelodySuggestions(measureIndex, melodySuggestions);
  }

  // 5. Play preview
  if (options.autoPreview) {
    await playChordWithVoicing(chord, bassVoicing);
  }
}
```

```javascript
// modules/ai/melodySuggestion.js

export function suggestMelodyNotes(measureIndex, chord, previousMelody, options) {
  const chordTones = getChordNotes(chord.root, chord.type);
  const scaleTones = getScaleNotes(chord.key);

  // Analyze previous melody
  const lastNote = previousMelody ? previousMelody.notes[previousMelody.notes.length - 1] : null;

  // Generate suggestions
  const suggestions = [];

  // 1. Chord tones (highest priority)
  chordTones.forEach(note => {
    suggestions.push({
      note: note,
      score: 95,
      reason: "Chord tone - strong and stable",
      category: "chord-tone"
    });
  });

  // 2. Stepwise motion from last note
  if (lastNote) {
    const stepwiseNotes = [
      getNextNote(lastNote, 1),   // step up
      getNextNote(lastNote, -1),  // step down
      getNextNote(lastNote, 2),   // whole step up
      getNextNote(lastNote, -2)   // whole step down
    ].filter(note => scaleTones.includes(note));

    stepwiseNotes.forEach(note => {
      suggestions.push({
        note: note,
        score: 85,
        reason: "Stepwise motion - smooth and melodic",
        category: "stepwise"
      });
    });
  }

  // 3. Approach tones (jazz style)
  if (options.style === 'jazz') {
    chordTones.forEach(target => {
      const approachTone = getNextNote(target, -1);  // chromatic approach
      suggestions.push({
        note: approachTone,
        score: 75,
        reason: `Chromatic approach to ${target}`,
        category: "approach-tone"
      });
    });
  }

  // 4. Passing tones
  scaleTones.forEach(note => {
    if (!chordTones.includes(note)) {
      suggestions.push({
        note: note,
        score: 65,
        reason: "Passing tone - adds movement",
        category: "passing-tone"
      });
    }
  });

  // Sort by score and remove duplicates
  return suggestions
    .sort((a, b) => b.score - a.score)
    .filter((note, index, self) =>
      index === self.findIndex(n => n.note === note.note)
    )
    .slice(0, 10);
}
```

```javascript
// modules/ai/autoHarmonize.js

export function suggestChordsForMelody(melodyNotes, key, options = {}) {
  const measures = groupIntoMeasures(melodyNotes);
  const suggestions = [];

  measures.forEach((measure, index) => {
    // Analyze which notes are in the measure
    const noteSet = new Set(measure.map(n => n.pitch % 12));

    // Find chords that contain most of these notes
    const candidates = findMatchingChords(noteSet, key);

    // Score each candidate
    const scored = candidates.map(chord => {
      const chordNotes = getChordNotes(chord.root, chord.type);
      const matchCount = countMatches(noteSet, chordNotes);
      const voiceLeadingScore = index > 0
        ? scoreVoiceLeading(suggestions[index - 1], chord)
        : 100;

      return {
        chord,
        score: (matchCount / noteSet.size) * 70 + voiceLeadingScore * 0.30,
        matchCount
      };
    });

    // Take top 3
    suggestions.push(scored.sort((a, b) => b.score - a.score).slice(0, 3));
  });

  return suggestions;
}
```

## Technical Implementation: Unified State Management

```javascript
// modules/state/compositionState.js

export class CompositionState {
  constructor() {
    this.measures = [];
    this.metadata = {
      title: "",
      tempo: 120,
      timeSignature: { num: 4, denom: 4 },
      key: "C"
    };

    // Event emitters for reactivity
    this.events = new EventEmitter();
  }

  // Chord Progression Methods
  addChord(measureIndex, chord) {
    this.measures[measureIndex].chord = chord;
    this.events.emit('chordChanged', measureIndex, chord);

    // Auto-update bass if it's auto-generated
    if (this.measures[measureIndex].notation.bass.autoGenerated) {
      this.updateBassFromChord(measureIndex);
    }
  }

  updateBassFromChord(measureIndex) {
    const chord = this.measures[measureIndex].chord;
    const prevChord = this.measures[measureIndex - 1]?.chord;

    const bassVoicing = generateBassVoicing(chord, prevChord);
    this.measures[measureIndex].notation.bass.voices[0] = bassVoicing;

    this.events.emit('bassUpdated', measureIndex, bassVoicing);
  }

  // Notation Methods
  updateMelodyNote(measureIndex, voiceIndex, noteIndex, newNote) {
    this.measures[measureIndex].notation.treble.voices[voiceIndex].notes[noteIndex] = newNote;
    this.events.emit('melodyChanged', measureIndex, voiceIndex, noteIndex);

    // Optionally trigger chord re-analysis
    if (this.settings.autoHarmonize) {
      this.analyzeChordFromMelody(measureIndex);
    }
  }

  // Integration Methods
  applyTemplate(template) {
    // Clear existing
    this.measures = [];

    // Apply template progression
    let measureIndex = 0;
    template.progressions.forEach(chordSpec => {
      for (let i = 0; i < chordSpec.measures; i++) {
        this.measures[measureIndex] = {
          number: measureIndex + 1,
          chord: {
            root: chordSpec.root,
            type: chordSpec.type,
            inversion: 0
          },
          notation: {
            treble: { voices: [{ notes: [] }] },
            bass: {
              voices: [{ notes: [] }],
              autoGenerated: true
            }
          }
        };

        // Auto-fill bass
        this.updateBassFromChord(measureIndex);

        measureIndex++;
      }
    });

    this.events.emit('templateApplied', template);
  }

  // Export Methods
  exportToMIDI() {
    // Use Tone.js MIDI export
    // Include both chord progression and notation
  }

  exportToMusicXML() {
    // Export full score with:
    // - Chord symbols
    // - Treble clef notation
    // - Bass clef notation
    // - Metadata
  }
}
```
